Campus Entry & Exit Monitoring for both Walking Pedestrians and Vehicles using Close-Range & Long-Range RFID.

2 RFID MODES ACTIVE
1. PEDESTRIANS 13.56 MHz / 125 kHz
Close-Range RFID Reader / ID Tag

Walking students, faculty, staff, and visitors tap or scan their ID card at pedestrian turnstiles and walkway readers upon entering or exiting campus.

Pedestrian Scan Flow:
Student approaches reader → Taps/scans RFID card → ESP32 reads UID → CHARRMPASS verifies user → ENTRY / EXIT Logged
2. VEHICLES 900 MHz UHF Long-Range
Long-Range RFID Reader / Windshield Tag

Motorcycles, cars, SUVs, and delivery vans with windshield/headlight UHF tags are automatically detected at a distance at boom barrier gates.

Vehicle Detection Flow:
Vehicle approaches gate → Long-range UHF detects tag → ESP32 receives tag → CHARRMPASS verifies vehicle → ENTRY / EXIT Logged

ESP32 Bluetooth Low Energy (BLE) Wi-Fi Provisioning Firmware Snippet (.ino)

Flash this production code snippet onto your ESP32 board to enable 100% Bluetooth Low Energy (BLE) Wi-Fi provisioning. No hardcoded credentials — SSID & password stored exclusively in ESP32 NVS Flash (Preferences). Credentials survive power cycles; send RESET via Serial Monitor to wipe and re-enter setup mode at any time.

#include <WiFi.h>
#include <HTTPClient.h>
#include <Preferences.h>
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

#define GATE_ID             "CHARRMPASS_GATE_ENTRY"
#define GATE_TYPE           "ENTRY"
#define BLE_DEVICE_NAME     "CHARRMPASS_ENTRY_BLE"
#define BLE_SERVICE_UUID    "4fafc201-1fb5-459e-8fcc-c5c9c331914b"
#define BLE_CHAR_UUID       "beb5483e-36e1-4688-b7f5-ea07361b26a8"

const char* SUPABASE_URL  = "https://sdwjkgtxrpeajuymgpxp.supabase.co";
const char* SUPABASE_ANON = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...";

Preferences preferences;
BLEServer* pBleServer = NULL;
BLECharacteristic* pBleCharacteristic = NULL;
bool bleClientConnected = false;
bool bleServerRunning = false;
bool newWifiCredentialsReceived = false;
String currentSsid = "";
String currentPass = "";

class BleServerCallbacks : public BLEServerCallbacks {
  void onConnect(BLEServer* pServer) {
    bleClientConnected = true;
    Serial.println("[BLE] Web Bluetooth Admin Connected!");
  }
  void onDisconnect(BLEServer* pServer) {
    bleClientConnected = false;
    Serial.println("[BLE] Bluetooth Client Disconnected.");
    if (bleServerRunning) BLEDevice::startAdvertising();
  }
};

class BleCharCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) override {
    String rxValue = pCharacteristic->getValue().c_str();
    if (rxValue.length() == 0) return;
    Serial.println("[BLE RX] " + rxValue);

    // Parse SSID and Password (JSON: {"ssid":"...","pass":"..."} or "SSID:PASS")
    String newSsid = "", newPass = "";
    int sIndex = rxValue.indexOf("\"ssid\"");
    if (sIndex != -1) {
      int c = rxValue.indexOf(':', sIndex);
      int q1 = rxValue.indexOf('"', c);
      int q2 = rxValue.indexOf('"', q1 + 1);
      newSsid = rxValue.substring(q1 + 1, q2);
    }
    int pIndex = rxValue.indexOf("\"pass\"");
    if (pIndex != -1) {
      int c = rxValue.indexOf(':', pIndex);
      int q1 = rxValue.indexOf('"', c);
      int q2 = rxValue.indexOf('"', q1 + 1);
      newPass = rxValue.substring(q1 + 1, q2);
    }
    if (newSsid.length() == 0) {
      int sep = rxValue.indexOf(':');
      if (sep != -1) {
        newSsid = rxValue.substring(0, sep);
        newPass = rxValue.substring(sep + 1);
      }
    }
    if (newSsid.length() > 0) {
      currentSsid = newSsid; currentPass = newPass;
      preferences.begin("charrm_wifi", false);
      preferences.putString("ssid", currentSsid);
      preferences.putString("pass", currentPass);
      preferences.end();
      
      // Notify Admin Phone / Web Browser of Saved Credentials
      if (pBleCharacteristic) {
        pBleCharacteristic->setValue("{\"event\":\"SAVED\",\"ssid\":\"" + currentSsid + "\"}");
        pBleCharacteristic->notify();
      }
      newWifiCredentialsReceived = true;
    }
  }
};

void sendDeviceHeartbeat() {
  if (WiFi.status() != WL_CONNECTED) return;
  HTTPClient http;
  String url = String(SUPABASE_URL) + "/rest/v1/devices?esp32_identifier=eq." + String(GATE_ID);
  http.begin(url);
  http.addHeader("apikey", SUPABASE_ANON);
  http.addHeader("Authorization", String("Bearer ") + SUPABASE_ANON);
  http.addHeader("Content-Type", "application/json");
  String payload = "{\"status\":\"ONLINE\",\"last_online\":\"now()\"}";
  int code = http.PATCH(payload);
  http.end();
}

void startBleServer() {
  if (bleServerRunning) return;
  BLEDevice::init(BLE_DEVICE_NAME);
  pBleServer = BLEDevice::createServer();
  pBleServer->setCallbacks(new BleServerCallbacks());
  BLEService *pService = pBleServer->createService(BLE_SERVICE_UUID);
  pBleCharacteristic = pService->createCharacteristic(
      BLE_CHAR_UUID,
      BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_WRITE | BLECharacteristic::PROPERTY_NOTIFY
  );
  pBleCharacteristic->setCallbacks(new BleCharCallbacks());
  pBleCharacteristic->addDescriptor(new BLE2902());
  pService->start();
  BLEAdvertising *pAdv = BLEDevice::getAdvertising();
  pAdv->addServiceUUID(BLE_SERVICE_UUID);
  pAdv->start();
  bleServerRunning = true;
}
